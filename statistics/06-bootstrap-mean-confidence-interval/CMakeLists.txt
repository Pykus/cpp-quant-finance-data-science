cmake_minimum_required(VERSION 3.16)
project(bootstrap_mean_ci LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(bootstrap_mean STATIC bootstrap_mean.cpp)
target_compile_options(bootstrap_mean PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(bootstrap_demo main.cpp)
target_link_libraries(bootstrap_demo PRIVATE bootstrap_mean)
add_executable(bootstrap_tests tests.cpp)
target_link_libraries(bootstrap_tests PRIVATE bootstrap_mean)
enable_testing()
add_test(NAME bootstrap_mean_tests COMMAND bootstrap_tests)
