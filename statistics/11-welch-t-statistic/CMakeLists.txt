cmake_minimum_required(VERSION 3.20)
project(welch_t_statistic LANGUAGES CXX)
include(CTest)
add_executable(welch_tests tests.cpp)
target_compile_features(welch_tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(welch_tests PRIVATE /W4 /WX)
else()
    target_compile_options(welch_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME welch_tests COMMAND welch_tests)
