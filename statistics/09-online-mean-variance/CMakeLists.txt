cmake_minimum_required(VERSION 3.20)
project(online_statistics LANGUAGES CXX)
include(CTest)
add_executable(online_statistics_tests tests.cpp)
target_compile_features(online_statistics_tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(online_statistics_tests PRIVATE /W4 /WX)
else()
    target_compile_options(online_statistics_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME online_statistics_tests COMMAND online_statistics_tests)
