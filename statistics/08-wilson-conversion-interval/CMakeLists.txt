cmake_minimum_required(VERSION 3.16)
project(wilson_conversion_interval LANGUAGES CXX)
add_executable(wilson_example main.cpp)
add_executable(wilson_tests tests.cpp)
target_compile_features(wilson_example PRIVATE cxx_std_20)
target_compile_features(wilson_tests PRIVATE cxx_std_20)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(wilson_example PRIVATE -Wall -Wextra -Werror -pedantic)
  target_compile_options(wilson_tests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
enable_testing()
add_test(NAME wilson_regression COMMAND wilson_tests)
