cmake_minimum_required(VERSION 3.16)
project(spearman_correlation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(spearman_demo main.cpp)
add_executable(spearman_tests tests.cpp)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(spearman_demo PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_compile_options(spearman_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME spearman_tests COMMAND spearman_tests)
