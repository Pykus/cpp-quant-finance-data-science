cmake_minimum_required(VERSION 3.20)
project(interpolated_quantiles LANGUAGES CXX)
include(CTest)
add_executable(quantile_tests tests.cpp)
target_compile_features(quantile_tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(quantile_tests PRIVATE /W4 /WX)
else()
    target_compile_options(quantile_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME interpolated_quantiles COMMAND quantile_tests)
