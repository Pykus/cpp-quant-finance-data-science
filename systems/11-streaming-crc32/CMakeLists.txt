cmake_minimum_required(VERSION 3.20)
project(streaming_crc32 LANGUAGES CXX)
include(CTest)
add_executable(streaming_crc32_tests tests.cpp)
target_compile_features(streaming_crc32_tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(streaming_crc32_tests PRIVATE /W4 /WX)
else()
    target_compile_options(streaming_crc32_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME streaming_crc32_tests COMMAND streaming_crc32_tests)
