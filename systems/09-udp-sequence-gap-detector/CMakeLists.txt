cmake_minimum_required(VERSION 3.20)
project(udp_sequence_gap_detector LANGUAGES CXX)
include(CTest)
add_executable(udp_sequence_gap_detector_tests tests.cpp)
target_compile_features(udp_sequence_gap_detector_tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(udp_sequence_gap_detector_tests PRIVATE /W4 /WX)
else()
    target_compile_options(udp_sequence_gap_detector_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME udp_sequence_gap_detector COMMAND udp_sequence_gap_detector_tests)
