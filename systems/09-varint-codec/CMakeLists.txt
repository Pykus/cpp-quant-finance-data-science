cmake_minimum_required(VERSION 3.16)
project(checked_varint_codec LANGUAGES CXX)
add_executable(varint_example main.cpp)
add_executable(varint_tests tests.cpp)
target_compile_features(varint_example PRIVATE cxx_std_20)
target_compile_features(varint_tests PRIVATE cxx_std_20)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(varint_example PRIVATE -Wall -Wextra -Werror -pedantic)
  target_compile_options(varint_tests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
enable_testing()
add_test(NAME varint_regression COMMAND varint_tests)
