cmake_minimum_required(VERSION 3.16)
project(spsc_market_event_queue LANGUAGES CXX)

find_package(Threads REQUIRED)

add_library(spsc_queue INTERFACE)
target_include_directories(spsc_queue INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(spsc_queue INTERFACE cxx_std_20)

foreach(target IN ITEMS queue_demo queue_tests queue_benchmark)
    if(target STREQUAL "queue_demo")
        set(source main.cpp)
    elseif(target STREQUAL "queue_tests")
        set(source tests.cpp)
    else()
        set(source benchmark.cpp)
    endif()
    add_executable(${target} ${source})
    target_link_libraries(${target} PRIVATE spsc_queue Threads::Threads)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()

enable_testing()
add_test(NAME spsc_queue_checks COMMAND queue_tests)
add_test(NAME spsc_queue_demo COMMAND queue_demo)
