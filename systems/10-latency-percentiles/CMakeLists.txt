cmake_minimum_required(VERSION 3.20)
project(latency_percentiles LANGUAGES CXX)
include(CTest)
add_executable(latency_percentiles_tests tests.cpp)
target_compile_features(latency_percentiles_tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(latency_percentiles_tests PRIVATE /W4 /WX)
else()
    target_compile_options(latency_percentiles_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME latency_percentiles_tests COMMAND latency_percentiles_tests)
