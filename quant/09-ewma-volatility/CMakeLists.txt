cmake_minimum_required(VERSION 3.16)
project(ewma_volatility LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(ewma INTERFACE)
target_include_directories(ewma INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(ewma_demo main.cpp)
target_link_libraries(ewma_demo PRIVATE ewma)
add_executable(ewma_tests tests.cpp)
target_link_libraries(ewma_tests PRIVATE ewma)
include(CTest)
if(BUILD_TESTING)
  add_test(NAME ewma_tests COMMAND ewma_tests)
endif()
