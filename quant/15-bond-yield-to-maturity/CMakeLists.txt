cmake_minimum_required(VERSION 3.20)
project(bond_yield_to_maturity LANGUAGES CXX)
include(CTest)
add_executable(bond_yield_tests tests.cpp)
target_compile_features(bond_yield_tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(bond_yield_tests PRIVATE /W4 /WX)
else()
    target_compile_options(bond_yield_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME bond_yield_tests COMMAND bond_yield_tests)
